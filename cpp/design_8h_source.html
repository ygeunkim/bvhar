101inline Eigen::MatrixXd build_ydummy(int p, const Eigen::VectorXd& sigma, double lambda,
102 const Eigen::VectorXd& daily, const Eigen::VectorXd& weekly, const Eigen::VectorXd& monthly,
103 bool include_mean) {
104 int dim = sigma.size();
105 Eigen::MatrixXd res = Eigen::MatrixXd::Zero(dim * p + dim + 1, dim); // Yp
106 // first block------------------------
107 res.block(0, 0, dim, dim).diagonal() = daily.array() * sigma.array() / lambda; // deltai * sigma or di * sigma
108 if (p > 1) {
109 // avoid error when p = 1
110 res.block(dim, 0, dim, dim).diagonal() = weekly.array() * sigma.array() / lambda; // wi * sigma
111 res.block(2 * dim, 0, dim, dim).diagonal() = monthly.array() * sigma.array() / lambda; // mi * sigma
112 }
113 // second block-----------------------
114 res.block(dim * p, 0, dim, dim).diagonal() = sigma;
115 if (include_mean) {
116 return res;
117 }
118 return res.topRows(dim * p + dim);
119}

121inline Eigen::MatrixXd build_xdummy(const Eigen::VectorXd& lag_seq,
122 double lambda, const Eigen::VectorXd& sigma,
123 double eps, bool include_mean) {
124 int dim = sigma.size();
125 int var_lag = lag_seq.size();
126 Eigen::MatrixXd Sig = Eigen::MatrixXd::Zero(dim, dim);
127 Eigen::MatrixXd res = Eigen::MatrixXd::Zero(dim * var_lag + dim + 1, dim * var_lag + 1);
128 // first block------------------
129 Eigen::MatrixXd Jp = Eigen::MatrixXd::Zero(var_lag, var_lag);
130 Jp.diagonal() = lag_seq;
131 Sig.diagonal() = sigma / lambda;
132 res.block(0, 0, dim * var_lag, dim * var_lag) = Eigen::kroneckerProduct(Jp, Sig);
133 // third block------------------
134 res(dim * var_lag + dim, dim * var_lag) = eps;
135 if (include_mean) {
136 return res;
137 }
138 return res.block(0, 0, dim * var_lag + dim, dim * var_lag);

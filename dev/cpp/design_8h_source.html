162inline Eigen::MatrixXd build_ydummy(int p, const Eigen::VectorXd& sigma, double lambda,
163 const Eigen::VectorXd& daily, const Eigen::VectorXd& weekly, const Eigen::VectorXd& monthly,
164 bool include_mean) {
165 int dim = sigma.size();
166 Eigen::MatrixXd res = Eigen::MatrixXd::Zero(dim * p + dim + 1, dim); // Yp
167 // first block------------------------
168 res.block(0, 0, dim, dim).diagonal() = daily.array() * sigma.array() / lambda; // deltai * sigma or di * sigma
169 if (p > 1) {
170 // avoid error when p = 1
171 res.block(dim, 0, dim, dim).diagonal() = weekly.array() * sigma.array() / lambda; // wi * sigma
172 res.block(2 * dim, 0, dim, dim).diagonal() = monthly.array() * sigma.array() / lambda; // mi * sigma
173 }
174 // second block-----------------------
175 res.block(dim * p, 0, dim, dim).diagonal() = sigma;
176 if (include_mean) {
177 return res;
178 }
179 return res.topRows(dim * p + dim);
180}

182inline Eigen::MatrixXd build_xdummy(const Eigen::VectorXd& lag_seq,
183 double lambda, const Eigen::VectorXd& sigma,
184 double eps, bool include_mean) {
185 int dim = sigma.size();
186 int var_lag = lag_seq.size();
187 Eigen::MatrixXd Sig = Eigen::MatrixXd::Zero(dim, dim);
188 Eigen::MatrixXd res = Eigen::MatrixXd::Zero(dim * var_lag + dim + 1, dim * var_lag + 1);
189 // first block------------------
190 Eigen::MatrixXd Jp = Eigen::MatrixXd::Zero(var_lag, var_lag);
191 Jp.diagonal() = lag_seq;
192 Sig.diagonal() = sigma / lambda;
193 res.block(0, 0, dim * var_lag, dim * var_lag) = Eigen::kroneckerProduct(Jp, Sig);
194 // third block------------------
195 res(dim * var_lag + dim, dim * var_lag) = eps;
196 if (include_mean) {
197 return res;
198 }
199 return res.block(0, 0, dim * var_lag + dim, dim * var_lag);
200}
